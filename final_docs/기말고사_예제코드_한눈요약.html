1. 출제 가능성 높은 예제코드

최우선

swap + calling convention

2023은 C convention, 2024는 STDCALL. ret와 ret 8 차이를 반드시 구분.

swap PROC
    push ebp
    mov  ebp, esp
    sub  esp, 4
    push eax
    push ebx
    push edx

    mov  eax, [ebp+8]     ; a
    mov  ebx, [ebp+12]    ; b
    mov  edx, [eax]
    mov  [ebp-4], edx     ; temp = *a
    mov  edx, [ebx]
    mov  [eax], edx       ; *a = *b
    mov  edx, [ebp-4]
    mov  [ebx], edx       ; *b = temp

    pop  edx
    pop  ebx
    pop  eax
    mov  esp, ebp
    pop  ebp
    ret  8                ; STDCALL
swap ENDP

; 호출
push OFFSET y
push OFFSET x
call swap
; C convention이면 여기서 add esp, 8

3. 조건 점프 표

    

      
        
의미UnsignedSigned조건

      
      
        
>JAJGJA: CF=0 AND ZF=0, JG: ZF=0 AND SF=OF

        
>=JAEJGEJAE: CF=0, JGE: SF=OF

        
<JBJLJB: CF=1, JL: SF!=OF

        
<=JBEJLEJBE: CF=1 OR ZF=1, JLE: ZF=1 OR SF!=OF

        
==JE/JZJE/JZZF=1

        
!=JNE/JNZJNE/JNZZF=0

      
    

    
기억법: Above/Below는 unsigned, Greater/Less는 signed. cmp left, right 다음 jump는 left ? right 기준으로 읽는다.

6. 스택 프레임과 호출 규약

    

      

        
표준 스택 프레임


MyProc PROC
    push ebp
    mov  ebp, esp
    sub  esp, 8

    mov  eax, [ebp+8]     ; 첫 번째 매개변수
    mov  ebx, [ebp+12]    ; 두 번째 매개변수
    mov  [ebp-4], eax     ; 지역 변수

    mov  esp, ebp
    pop  ebp
    ret  8                ; STDCALL
MyProc ENDP

      

      

        
STDCALL swap


swap PROC
    push ebp
    mov  ebp, esp
    sub  esp, 4
    push eax
    push ebx
    push edx

    mov  eax, [ebp+8]
    mov  ebx, [ebp+12]
    mov  edx, [eax]
    mov  [ebp-4], edx
    mov  edx, [ebx]
    mov  [eax], edx
    mov  edx, [ebp-4]
    mov  [ebx], edx

    pop  edx
    pop  ebx
    pop  eax
    mov  esp, ebp
    pop  ebp
    ret  8
swap ENDP

      

    


    

      
구분인자 push 순서스택 정리호출 예

      
        
C convention오른쪽 인자부터caller가 add esp, npush OFFSET y, push OFFSET x, call swap, add esp,8

        
STDCALL오른쪽 인자부터callee가 ret npush OFFSET y, push OFFSET x, call swap

REAL4 해석


; IEEE 754 single precision
; sign 1비트, exponent 8비트, fraction 23비트
; exponent bias = 127

; 예: 3FC00000h
; sign=0, exponent=127, fraction=.5
; 값 = 1.5 * 2^0 = 1.5

; 메모리 저장은 little endian
; 3FC00000h -> 00 00 C0 3F

      

      

        
기출 실행 결과 추적

sar al, 2       ; E0h, signed shift

; 2023: IMUL

mov bl, 6
imul bl         ; AX = 008Ah, OF=1

; 2024: LOOPNE

L1: inc al

  • REAL4 해석 절차: sign, exponent-bias, fraction, little endian.

3회전: 변형 대응

        
기출과 숫자, 부등호, calling convention만 바뀌는 문제를 대비한다.


; 변형 1: signed AND
if (A > B && C <= D) X = 1; else X = 0;
; 첫 조건의 반대: jle FALSE
; 둘째 조건의 반대: jg FALSE

; 변형 2: unsigned OR
if (A < B || C >= D) X = 1;
; 첫 조건 참: jb TRUE
; 둘째 조건 참: jae TRUE

; 변형 3: signed remainder
D = (A - B * 2) % C
; EAX에 피제수, CDQ, IDIV C, EDX 저장
